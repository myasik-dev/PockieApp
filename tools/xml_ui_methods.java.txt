  private void home() {
    View page=showLayout(R.layout.screen_home);
    action(page,R.id.open_settings,()->go("Settings"));
    action(page,R.id.tracking_status,this::trackingStatus);
    action(page,R.id.see_all,()->go("Transactions"));
    List<Transaction> all=store.all();
    boolean demo=!all.isEmpty()&&all.stream().allMatch(t->t.demo);
    Calendar month=Calendar.getInstance();if(demo)month.setTimeInMillis(all.get(0).date);
    long income=0,expense=0;
    for(Transaction t:all)if(sameMonth(t.date,month)&&included(t)){if(t.income)income+=t.amountPaise;else expense+=t.amountPaise;}
    bindText(page,R.id.home_month,demo?"October · Sample data":"This Month");
    bindText(page,R.id.home_spending,money(expense));bindText(page,R.id.home_income,money(income));
    bindText(page,R.id.home_average,money(expense/Math.max(1,month.get(Calendar.DAY_OF_MONTH))));
    bindText(page,R.id.home_expense,money(expense));
    page.findViewById(R.id.home_recent).setMinimumHeight(Math.max(d(200),getResources().getDisplayMetrics().heightPixels-d(420)));
    LinearLayout list=page.findViewById(R.id.recent_transactions);
    if(all.isEmpty())empty(list,"Your money, beautifully organized","Add your first expense or scan your bank SMS.",true);
    else for(int i=0;i<Math.min(8,all.size());i++)transactionRow(list,all.get(i),false,false);
    page.findViewById(R.id.clear_demo_home).setVisibility(demo?View.VISIBLE:View.GONE);
    action(page,R.id.clear_demo_home,this::clearDemo);
  }

  private void transactionRow(ViewGroup parent,Transaction t,boolean signed,boolean detail) {
    View row=getLayoutInflater().inflate(R.layout.item_transaction,parent,false);
    bindTransactionRow(row,t,signed,detail);parent.addView(row);
  }

  private void bindTransactionRow(View row,Transaction t,boolean signed,boolean detail) {
    boolean godaddy=t.merchant.toLowerCase(Locale.ROOT).contains("godaddy");
    String icon=t.category.equals("Food")?"food":t.category.equals("Transfers")?"person":godaddy?"cart":"shop";
    int color=t.category.equals("Food")?0xffff7a00:t.category.equals("Transfers")?(t.merchant.startsWith("Sarath")?0xffffad30:0xff35b768):godaddy?BLUE:0xff8040ff;
    row.findViewById(R.id.transaction_badge).setBackgroundTintList(ColorStateList.valueOf(color));
    ((PockieIcon)row.findViewById(R.id.transaction_icon)).setIconName(icon);
    bindText(row,R.id.transaction_merchant,t.merchant);
    String subtitle=detail?date(t.date,"dd MMM yyyy, hh:mm a"):signed?(t.source.equals("SMS")||t.source.equals("Notification")||t.demo?"UPI • "+(t.reference.isEmpty()?t.account:t.reference):t.source+" • "+t.category):date(t.date,"dd MMM yyyy");
    bindText(row,R.id.transaction_subtitle,subtitle);
    TextView amount=row.findViewById(R.id.transaction_amount);amount.setText((signed?(t.income?"+ ":"- "):"")+money(t.amountPaise));
    amount.setTextColor(signed?(t.income?GREEN:RED):BLACK);
    row.setOnClickListener(detail?null:v->details(t));
    row.setContentDescription(t.merchant+", "+(t.income?"income ":"expense ")+money(t.amountPaise));
  }

  private void transactions() {
    View page=showLayout(R.layout.screen_transactions);
    toolbar(page,categoryFilter.isEmpty()?"Transactions":categoryFilter+" Transactions",true);
    TextView category=page.findViewById(R.id.category_filter_chip);
    category.setVisibility(categoryFilter.isEmpty()?View.GONE:View.VISIBLE);category.setText("Category: "+categoryFilter+"   ×");
    category.setOnClickListener(v->{categoryFilter="";render();});
    TextView searched=page.findViewById(R.id.search_filter_chip);searched.setVisibility(query.isEmpty()?View.GONE:View.VISIBLE);searched.setText("Search: "+query+"   ×");searched.setOnClickListener(v->{query="";render();});
    int[] ids={R.id.filter_all,R.id.filter_expense,R.id.filter_income};String[] names={"All","Expense","Income"};
    for(int i=0;i<ids.length;i++){String name=names[i];TextView tab=page.findViewById(ids[i]);boolean selected=filter.equals(name);tab.setBackgroundResource(selected?R.drawable.bg_pill_black:R.drawable.bg_pill_outline);tab.setTextColor(selected?Color.WHITE:0xffb7b7b7);tab.setOnClickListener(v->{filter=name;render();});}
    LinearLayout list=page.findViewById(R.id.transaction_list);String last="";int count=0;
    for(Transaction t:store.all()) {
      if((filter.equals("Expense")&&t.income)||(filter.equals("Income")&&!t.income)||(!categoryFilter.isEmpty()&&!categoryFilter.equals(t.category)))continue;
      if(!query.isEmpty()&&!(t.merchant+" "+t.reference+" "+t.category+" "+t.account+" "+t.notes).toLowerCase(Locale.ROOT).contains(query.toLowerCase(Locale.ROOT)))continue;
      String day=date(t.date,"dd MMMM yyyy");
      if(!day.equals(last)){
        TextView header=(TextView)getLayoutInflater().inflate(R.layout.item_date_header,list,false);header.setText(day);
        if(count>0)((LinearLayout.LayoutParams)header.getLayoutParams()).topMargin=d(18);
        list.addView(header);last=day;
      }
      transactionRow(list,t,true,false);count++;
    }
    if(count==0)empty(list,"No transactions found",query.isEmpty()?"Your transactions will appear here.":"Try a different merchant, category, or reference.",false);
  }

  private void details(Transaction t) {
    Sheet sheet=new Sheet(R.layout.bottom_sheet_transaction_details,"Transaction Details","Choose a quick way to add your income or\nexpense");
    bindTransactionRow(sheet.view.findViewById(R.id.detail_transaction_summary),t,true,true);
    LinearLayout fields=sheet.view.findViewById(R.id.transaction_detail_fields);
    detailLine(fields,"Type",t.income?"Income":"Expense",t.income?GREEN:RED,t.income?0xffddf8e8:0xffffe2e4);
    detailLine(fields,"Category",t.category,BLACK,colorFor(t.category));
    detailLine(fields,"Account",t.account.isEmpty()?"Cash / not specified":t.account,BLACK,0);
    detailLine(fields,"Reference No",t.reference.isEmpty()?t.source+"/"+t.id:t.reference,BLACK,0);
    if(!t.notes.isEmpty())detailLine(fields,"Notes",t.notes,BLACK,0);
    action(sheet.view,R.id.edit_transaction,()->editor(t,sheet.dialog));
    action(sheet.view,R.id.delete_transaction,()->deleteDialog(t,sheet.dialog));sheet.dialog.show();
  }

  private void messageDialog(String title,String message,String negative,Runnable cancelled,String positive,Runnable confirmed){

  private void editor(Transaction original,BottomSheetDialog details) {
    boolean editing=original!=null&&original.id>0;Transaction t=original==null?new Transaction():original.copy();
    Sheet sheet=new Sheet(editing?R.layout.bottom_sheet_edit_transaction:R.layout.bottom_sheet_add_transaction,editing?"Edit "+(t.income?"Income":"Expense"):"Add Expense & Income","Choose a quick way to add your income or\nexpense");
    View view=sheet.view;
    if(!editing){
      TextView expense=view.findViewById(R.id.type_expense),income=view.findViewById(R.id.type_income);
      Runnable paint=()->{expense.setBackgroundResource(t.income?android.R.color.transparent:R.drawable.bg_pill_blue);expense.setTextColor(t.income?BLACK:Color.WHITE);income.setBackgroundResource(t.income?R.drawable.bg_pill_blue:android.R.color.transparent);income.setTextColor(t.income?Color.WHITE:BLACK);};
      expense.setOnClickListener(v->{t.income=false;paint.run();});income.setOnClickListener(v->{t.income=true;paint.run();});paint.run();
    }
    EditText merchant=view.findViewById(R.id.merchant_input),amount=view.findViewById(R.id.amount_input),notes=view.findViewById(R.id.notes_input);
    EditText account=view.findViewById(R.id.account_input),reference=view.findViewById(R.id.reference_input);
    merchant.setText(t.merchant);amount.setText(t.amountPaise==0?"":BigDecimal.valueOf(t.amountPaise,2).stripTrailingZeros().toPlainString());notes.setText(t.notes);
    if(account!=null)account.setText(t.account);if(reference!=null)reference.setText(t.reference);
    TextView selectedDate=view.findViewById(R.id.selected_date);selectedDate.setText(t.date>0?date(t.date,"dd MMM yyyy"):"Date");selectedDate.setTextColor(t.date>0?BLACK:0xffa6a8ae);
    action(view,R.id.date_selector,()->{
      Calendar initial=Calendar.getInstance();initial.setTimeInMillis(t.date>0?t.date:System.currentTimeMillis());
      new DatePickerDialog(this,(picker,year,month,day)->{Calendar chosen=Calendar.getInstance();chosen.setTimeInMillis(t.date>0?t.date:System.currentTimeMillis());chosen.set(year,month,day);t.date=chosen.getTimeInMillis();selectedDate.setText(date(t.date,"dd MMM yyyy"));selectedDate.setTextColor(BLACK);},initial.get(Calendar.YEAR),initial.get(Calendar.MONTH),initial.get(Calendar.DAY_OF_MONTH)).show();
    });
    TextView selectedCategory=view.findViewById(R.id.selected_category);selectedCategory.setText(t.category);
    action(view,R.id.category_selector,()->new AlertDialog.Builder(this).setTitle("Choose category").setAdapter(new ArrayAdapter<>(this,R.layout.item_dialog_choice,R.id.choice_text,categories),(dialog,which)->{t.category=categories[which];selectedCategory.setText(t.category);}).show());
    TextView hint=view.findViewById(R.id.receipt_review_hint);hint.setVisibility(t.source.equals("Receipt")?View.VISIBLE:View.GONE);
    if(t.source.equals("Receipt"))hint.setText(t.amountPaise==0?"Couldn’t identify the receipt total. Enter the amount and review the other fields before saving.":"Review the scanned merchant, total, and date before saving.");
    action(view,R.id.cancel_transaction,sheet.dialog::dismiss);
    action(view,R.id.save_transaction,()->{
      String name=merchant.getText().toString().trim();long value=BankParser.paise(amount.getText().toString().trim());
      if(name.isEmpty()){merchant.setError("Enter a merchant or payee");merchant.requestFocus();return;}
      if(value<=0||value>999999999999L){amount.setError("Enter a valid amount with up to 2 decimal places");amount.requestFocus();return;}
      if(t.date<=0){toast("Choose the receipt date before saving.");return;}
      t.merchant=name;t.amountPaise=value;t.notes=notes.getText().toString().trim();if(reference!=null)t.reference=reference.getText().toString().trim();if(account!=null)t.account=account.getText().toString().trim();
      if(store.save(t)<=0){toast("This transaction already exists.");return;}
      ((InputMethodManager)getSystemService(INPUT_METHOD_SERVICE)).hideSoftInputFromWindow(merchant.getWindowToken(),0);
      sheet.dialog.dismiss();if(details!=null){details.dismiss();details(t);}render();toast(editing?"Transaction updated":"Transaction saved");
    });
    sheet.dialog.show();
  }

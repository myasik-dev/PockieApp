  @Override
  public void onCreate(Bundle state) {
    super.onCreate(state);
    store = new ExpenseStore(this);
    prefs = Tracking.prefs(this);
    getOnBackPressedDispatcher()
        .addCallback(
            this,
            new androidx.activity.OnBackPressedCallback(true) {
              @Override
              public void handleOnBackPressed() {
                if (onboarding == 1) {
                  onboarding = 0;
                  render();
                } else if (!screen.equals("Home")) {
                  go("Home");
                } else {
                  setEnabled(false);
                  getOnBackPressedDispatcher().onBackPressed();
                  setEnabled(true);
                }
              }
            });
    getWindow()
        .setFlags(

    androidx.core.view.WindowCompat.setDecorFitsSystemWindows(getWindow(), false);
    getWindow().setNavigationBarColor(Color.WHITE);

      getWindow().setAttributes(attributes);

          v.setPadding(bars.left, 0, bars.right, Math.max(bars.bottom, ime.bottom));

  @Override

    super.onSaveInstanceState(state);
    state.putString("screen", screen);

    state.putString("query", query);

  @Override
  protected void onStart() {
    super.onStart();
    IntentFilter f = new IntentFilter(ExpenseStore.CHANGED);
    androidx.core.content.ContextCompat.registerReceiver(
        this, refresh, f, androidx.core.content.ContextCompat.RECEIVER_NOT_EXPORTED);
    registered = true;
  }

  @Override
  protected void onResume() {
    super.onResume();
    if (root != null && onboarding == 2 && screen.equals("Settings")) render();
  }

  @Override
  protected void onStop() {
    super.onStop();
    if (registered) unregisterReceiver(refresh);
    registered = false;
  }

  @Override
  protected void onDestroy() {
    worker.shutdown();
    store.close();
    super.onDestroy();
  }

  private FrameLayout.LayoutParams fp(

    p.setMargins(d(left), d(top), d(right), d(bottom));

  private int artFor(String category) {
    switch (category) {
      case "Food":

      case "Fuel":

      case "Business":
        return R.drawable.category_business;

      case "Transfers":

  private int colorFor(String category) {
    for (int i = 0; i < categories.length; i++)
      if (categories[i].equals(category)) return categoryColors[i];
    return categoryColors[2];
  }

  private String money(long paise) {
    NumberFormat nf = NumberFormat.getNumberInstance(new Locale("en", "IN"));
    nf.setMaximumFractionDigits(paise % 100 == 0 ? 0 : 2);
    nf.setMinimumFractionDigits(paise % 100 == 0 ? 0 : 2);
    return "₹ " + nf.format(BigDecimal.valueOf(paise, 2));
  }

  private String date(long timestamp, String pattern) {
    return new SimpleDateFormat(pattern, Locale.ENGLISH).format(new Date(timestamp));
  }

  private boolean included(Transaction t) {
    return !prefs.getBoolean("exclude_transfer", true) || !t.category.equals("Transfers");
  }

  private boolean sameMonth(long timestamp, Calendar month) {
    Calendar c = Calendar.getInstance();
    c.setTimeInMillis(timestamp);
    return c.get(Calendar.YEAR) == month.get(Calendar.YEAR)
        && c.get(Calendar.MONTH) == month.get(Calendar.MONTH);
  }

  private void render() {
    if (isFinishing()) return;
    root.removeAllViews();
    if (onboarding < 2) {
      renderOnboarding();
      return;
    }
    body = new FrameLayout(this);
    boolean hasNav = !screen.equals("Settings");
    root.addView(body, fp(-1, -1, Gravity.TOP, 0, 0, 0, hasNav ? 70 : 0));
    switch (screen) {
      case "Transactions":
        transactions();
        break;
      case "Categories":
        categoryScreen();
        break;
      case "Reports":
        reports();
        break;
      case "Settings":
        settings();
        break;
      default:
        home();
    }
    if (hasNav) bottomNav();
  }

  private void finishOnboarding() {
    prefs.edit().putBoolean("onboarded", true).apply();
    onboarding = 2;
    render();
  }

  private void requestSms(boolean onboard) {
    if (checkSelfPermission(Manifest.permission.READ_SMS) == PackageManager.PERMISSION_GRANTED
        && checkSelfPermission(Manifest.permission.RECEIVE_SMS)
            == PackageManager.PERMISSION_GRANTED) {
      prefs.edit().putBoolean("sms", true).apply();
      if (onboard) finishOnboarding();
      else render();
      scanSms();
      return;
    }
    new AlertDialog.Builder(this)
        .setTitle("Automatically track bank payments")
        .setMessage(
            "Pockie reads bank transaction messages on this phone to create income and expense"
                + " records. Messages stay on your device. You can turn this off in Settings, or"
                + " continue with manual entry.")
        .setPositiveButton(
            "Allow access",
            (dialog, which) ->
                requestPermissions(
                    new String[] {Manifest.permission.READ_SMS, Manifest.permission.RECEIVE_SMS},
                    onboard ? 101 : 102))
        .setNegativeButton(
            "Not now",
            (dialog, which) -> {
              if (onboard) finishOnboarding();
              else render();
            })
        .show();
  }

  @Override
  public void onRequestPermissionsResult(int request, String[] permissions, int[] results) {
    super.onRequestPermissionsResult(request, permissions, results);
    if (request == 101 || request == 102) {
      boolean granted =
          checkSelfPermission(Manifest.permission.READ_SMS) == PackageManager.PERMISSION_GRANTED;
      boolean receive =
          checkSelfPermission(Manifest.permission.RECEIVE_SMS) == PackageManager.PERMISSION_GRANTED;
      prefs.edit().putBoolean("sms", granted && receive).apply();
      if (request == 101) finishOnboarding();
      else render();
      if (granted) scanSms();
      else toast("SMS access is off. You can still add expenses manually.");
    }
  }

  private void transactions() {
    LinearLayout p =
        page(categoryFilter.isEmpty() ? "Transactions" : categoryFilter + " Transactions", true);
    LinearLayout content = column(this);
    pad(content, 18, 0, 18, 20);
    add(p, content, -2);
    if (!categoryFilter.isEmpty()) {
      TextView chip = label("Category: " + categoryFilter + "   ×", 12, BLUE, false);
      pad(chip, 10, 7, 10, 7);
      chip.setBackground(outline(0xffeaf5ff, 20, 0xffc7dfff, this));
      chip.setOnClickListener(
          v -> {
            categoryFilter = "";
            render();
          });
      content.addView(chip, lp(-2, 32));
      space(content, 14);
    }
    if (!query.isEmpty()) {
      TextView searched = label("Search: " + query + "   ×", 13, MUTED, false);
      searched.setOnClickListener(
          v -> {
            query = "";
            render();
          });
      add(content, searched, 32);
    }
    LinearLayout tabs = row(this);
    String[] filters = {"All", "Expense", "Income"};
    for (int i = 0; i < 3; i++) {
      String name = filters[i];
      TextView b = label(name, 15, filter.equals(name) ? Color.WHITE : 0xffb7b7b7, true);
      b.setGravity(Gravity.CENTER);
      b.setBackground(
          filter.equals(name) ? bg(BLACK, 24, this) : outline(Color.WHITE, 24, 0xffd4d4d4, this));
      LinearLayout.LayoutParams bp = new LinearLayout.LayoutParams(0, d(40), 1);
      if (i < 2) bp.rightMargin = d(16);
      tabs.addView(b, bp);
      b.setOnClickListener(
          v -> {
            filter = name;
            render();
          });
    }
    add(content, tabs, 40);
    space(content, 18);
    LinearLayout list = column(this);
    pad(list, 18, 0, 18, 16);
    ScrollView sc = scroll(list);
    p.addView(sc, new LinearLayout.LayoutParams(-1, 0, 1));
    String lastDate = "";
    int count = 0;
    for (Transaction t : store.all()) {
      if ((filter.equals("Expense") && t.income)
          || (filter.equals("Income") && !t.income)
          || (!categoryFilter.isEmpty() && !categoryFilter.equals(t.category))) continue;
      if (!query.isEmpty()
          && !(t.merchant + " " + t.reference + " " + t.category + " " + t.account + " " + t.notes)
              .toLowerCase(Locale.ROOT)
              .contains(query.toLowerCase(Locale.ROOT))) continue;
      String group = date(t.date, "dd MMMM yyyy");
      if (!group.equals(lastDate)) {
        if (count > 0) space(list, 18);
        TextView day = label(group, 15, 0xff5b5c69, true);
        add(list, day, 40);
        lastDate = group;
      }
      add(list, transactionRow(t, true), 61);
      count++;
    }
    if (count == 0)
      empty(
          list,
          "No transactions found",
          query.isEmpty()
              ? "Your transactions will appear here."
              : "Try a different merchant, category, or reference.",
          false);
  }

                  (view, y, m, day) -> {
                    Calendar pick = Calendar.getInstance();
                    pick.setTimeInMillis(t.date > 0 ? t.date : System.currentTimeMillis());

    pad(cat, 13, 0, 12, 0);
    cat.setBackground(bg(FIELD, 12, this));
    cat.addView(new Icon(this, "category", 0xff666886), lp(20, 20));
    TextView selectedCat = label(t.category, 14, BLACK, true);
    pad(selectedCat, 12, 0, 0, 0);
    cat.addView(selectedCat, new LinearLayout.LayoutParams(0, -2, 1));
    cat.addView(new Icon(this, "down", BLACK), lp(16, 16));
    add(p, cat, 46);
    cat.setOnClickListener(
        v ->
            new AlertDialog.Builder(this)
                .setTitle("Choose category")
                .setItems(
                    categories,
                    (dd, which) -> {
                      t.category = categories[which];
                      selectedCat.setText(t.category);
                    })
                .show());
    cat.setContentDescription("Choose category");
    space(p, 11);
    EditText account = null, reference = null;
    if (editing) {
      account =
          field(
              p,
              "bank",
              "Account (optional)",
              t.account,
              46,
              android.text.InputType.TYPE_CLASS_TEXT);
      space(p, 11);
    } else {
      reference =
          field(
              p,
              "notes",
              "Reference No (optional)",
              t.reference,
              46,
              android.text.InputType.TYPE_CLASS_TEXT);
      space(p, 11);
    }
    EditText notes =
        field(
            p,
            "notes",
            "Notes (optional)",
            t.notes,
            72,
            android.text.InputType.TYPE_CLASS_TEXT
                | android.text.InputType.TYPE_TEXT_FLAG_MULTI_LINE);
    space(p, 16);
    if (t.source.equals("Receipt")) {
      TextView hint =
          label(
              t.amountPaise == 0
                  ? "Couldn’t identify the receipt total. Enter the amount and review the other"
                      + " fields before saving."
                  : "Review the scanned merchant, total, and date before saving.",
              12,
              MUTED,
              false);
      add(p, hint, -2);
      space(p, 12);
    }
    LinearLayout buttons = row(this);
    TextView cancel = button(this, "Cancel", Color.WHITE, BLACK, false, dialog::dismiss);
    cancel.setBackground(outline(Color.WHITE, 12, 0xffdce0e8, this));
    EditText accountField = account;
    EditText referenceField = reference;
    TextView save =
        button(
            this,
            "Save",
            BLACK,
            Color.WHITE,
            false,
            () -> {
              String name = merchant.getText().toString().trim(),
                  value = amount.getText().toString().trim();
              long amountValue = BankParser.paise(value);
              if (name.isEmpty()) {
                merchant.setError("Enter a merchant or payee");
                merchant.requestFocus();
                return;
              }
              if (amountValue <= 0 || amountValue > 999999999999L) {
                amount.setError("Enter a valid amount with up to 2 decimal places");
                amount.requestFocus();
                return;
              }
              t.merchant = name;
              if (t.date <= 0) {
                toast("Choose the receipt date before saving.");
                return;
              }
              t.amountPaise = amountValue;
              if (referenceField != null) t.reference = referenceField.getText().toString().trim();
              t.notes = notes.getText().toString().trim();
              if (accountField != null) t.account = accountField.getText().toString().trim();
              if (store.save(t) <= 0) {
                toast("This transaction already exists.");
                return;
              }
              ((InputMethodManager) getSystemService(INPUT_METHOD_SERVICE))
                  .hideSoftInputFromWindow(saveToken(merchant), 0);
              dialog.dismiss();
              if (details != null) {
                details.dismiss();
                details(t);
              }
              render();
              toast(editing ? "Transaction updated" : "Transaction saved");
            });
    LinearLayout.LayoutParams bp = new LinearLayout.LayoutParams(0, d(46), 1);
    bp.rightMargin = d(12);
    buttons.addView(cancel, bp);
    buttons.addView(save, new LinearLayout.LayoutParams(0, d(46), 1));
    add(p, buttons, 46);
    dialog.show();
  }

  }

  private void scanSms() {
    if (scanning) return;
    if (checkSelfPermission(Manifest.permission.READ_SMS) != PackageManager.PERMISSION_GRANTED) {
      requestSms(false);
      return;
    }
    scanning = true;
    ProgressDialog progress =
        ProgressDialog.show(this, "Scan SMS", "Looking for bank transactions…", true, false);
    worker.execute(
        () -> {
          int found = 0;
          try (Cursor cursor =
              getContentResolver()
                  .query(
                      Telephony.Sms.Inbox.CONTENT_URI,
                      new String[] {"address", "body", "date"},
                      null,
                      null,
                      "date DESC")) {
            if (cursor != null)
              while (cursor.moveToNext()) {
                if (Tracking.importMessage(
                    this, cursor.getString(0), cursor.getString(1), cursor.getLong(2), "SMS"))
                  found++;
              }
            int count = found;
            runOnUiThread(
                () -> {
                  scanning = false;
                  if (!isFinishing()) {
                    progress.dismiss();
                    render();
                    toast(
                        count == 0
                            ? "No new bank transactions found."
                            : count + " bank transactions added.");
                  }
                });
          } catch (Exception e) {
            runOnUiThread(
                () -> {
                  scanning = false;
                  if (!isFinishing()) {
                    progress.dismiss();
                    toast("Unable to scan SMS. Check SMS access in Android Settings.");
                  }
                });
          }
        });
  }

  private String backup() throws JSONException {
    JSONObject root = new JSONObject();
    root.put("version", 1);
    JSONArray items = new JSONArray();
    for (Transaction t : store.all()) {
      JSONObject o = new JSONObject();
      o.put("merchant", t.merchant);
      o.put("amountPaise", t.amountPaise);
      o.put("date", t.date);
      o.put("income", t.income);
      o.put("category", t.category);
      o.put("account", t.account);
      o.put("reference", t.reference);
      o.put("notes", t.notes);
      o.put("source", t.source);
      o.put("demo", t.demo);
      o.put(
          "fingerprint",
          t.fingerprint.isEmpty()
              ? "backup:" + t.date + ":" + t.amountPaise + ":" + t.income + ":" + t.merchant
              : t.fingerprint);
      items.put(o);
    }
    root.put("transactions", items);
    return root.toString(2);
  }

  private void export(Uri uri) {
    try {
      String output = pendingExport.equals("csv") ? ExpenseStore.csv(store.all()) : backup();
      try (OutputStream stream = getContentResolver().openOutputStream(uri, "wt")) {
        if (stream == null) throw new IOException();
        stream.write(output.getBytes(java.nio.charset.StandardCharsets.UTF_8));
      }
      toast("Export saved.");
    } catch (Exception e) {
      toast("Unable to save export. Try another location.");
    }
  }

  private void restore(Uri uri) {
    worker.execute(
        () -> {
          try (InputStream input = getContentResolver().openInputStream(uri)) {
            if (input == null) throw new IOException();
            ByteArrayOutputStream bytes = new ByteArrayOutputStream();
            byte[] chunk = new byte[8192];
            int read;
            while ((read = input.read(chunk)) != -1) {
              if (bytes.size() + read > 10 * 1024 * 1024) throw new IOException();
              bytes.write(chunk, 0, read);
            }
            JSONObject root = new JSONObject(bytes.toString("UTF-8"));
            if (root.getInt("version") != 1) throw new JSONException("Unsupported version");
            JSONArray rows = root.getJSONArray("transactions");
            List<Transaction> drafts = new ArrayList<>();
            for (int i = 0; i < rows.length(); i++) {
              JSONObject o = rows.getJSONObject(i);
              Transaction t = new Transaction();
              t.merchant = o.getString("merchant");
              t.amountPaise = o.getLong("amountPaise");
              t.date = o.getLong("date");
              t.income = o.getBoolean("income");
              t.category = o.getString("category");
              t.account = o.optString("account");
              t.reference = o.optString("reference");
              t.notes = o.optString("notes");
              t.source = o.optString("source", "Backup");
              t.demo = o.optBoolean("demo");
              t.fingerprint = o.getString("fingerprint");
              if (t.merchant.trim().isEmpty()
                  || t.amountPaise <= 0
                  || t.amountPaise > 999999999999L
                  || t.date <= 0
                  || !Arrays.asList(categories).contains(t.category)
                  || t.fingerprint.isEmpty()) throw new JSONException("Invalid transaction");
              drafts.add(t);
            }
            runOnUiThread(
                () ->
                    new AlertDialog.Builder(this)
                        .setTitle("Restore backup?")
                        .setMessage(
                            "Merge "
                                + drafts.size()
                                + " transactions with your current records? Existing records are"
                                + " kept and matching backup records are skipped.")
                        .setNegativeButton("Cancel", null)
                        .setPositiveButton(
                            "Restore",
                            (v, w) -> {
                              int added = 0;
                              android.database.sqlite.SQLiteDatabase db =
                                  store.getWritableDatabase();
                              db.beginTransaction();
                              try {
                                for (Transaction t : drafts) if (store.save(t) > 0) added++;
                                db.setTransactionSuccessful();
                              } finally {
                                db.endTransaction();
                              }
                              render();
                              toast(added + " transactions restored.");
                            })
                        .show());
          } catch (Exception e) {
            runOnUiThread(
                () -> toast("This is not a valid Pockie backup. No records were changed."));
          }
        });
  }

  @Override
  protected void onActivityResult(int request, int result, Intent data) {
    super.onActivityResult(request, result, data);
    if (result != RESULT_OK) return;
    if (request == 202 && cameraUri != null) recognize(cameraUri);
    else if (data != null && data.getData() != null) {
      Uri uri = data.getData();
      if (request == 201) recognize(uri);
      else if (request == 301) export(uri);
      else if (request == 302) restore(uri);
    }
  }

  private void reports() {
    LinearLayout p = page("Reports", false);
    LinearLayout content = column(this);
    pad(content, 20, 12, 20, 28);
    add(p, scroll(content), -1);
    LinearLayout months = row(this);
    months.addView(
        iconButton(
            "back",
            BLACK,
            "Previous month",
            () -> {
              reportMonth.add(Calendar.MONTH, -1);
              render();
            }),
        lp(42, 42));
    TextView month = label(date(reportMonth.getTimeInMillis(), "MMMM yyyy"), 17, BLACK, true);
    month.setGravity(Gravity.CENTER);
    months.addView(month, new LinearLayout.LayoutParams(0, d(42), 1));
    months.addView(
        iconButton(
            "next",
            BLACK,
            "Next month",
            () -> {
              reportMonth.add(Calendar.MONTH, 1);
              render();
            }),
        lp(42, 42));
    add(content, months, 48);
    space(content, 18);
    long expense = 0, income = 0;
    Map<String, Long> sums = new LinkedHashMap<>();
    for (String c : categories) sums.put(c, 0L);
    for (Transaction t : store.all())
      if (sameMonth(t.date, reportMonth) && included(t)) {
        if (t.income) income += t.amountPaise;
        else {
          expense += t.amountPaise;
          sums.put(t.category, sums.getOrDefault(t.category, 0L) + t.amountPaise);
        }
      }
    LinearLayout total = column(this);
    pad(total, 18, 18, 18, 18);
    total.setBackground(bg(0xffffe675, 18, this));
    add(total, label("Total Spending", 15, BLACK, true), -2);
    space(total, 8);
    add(total, label(money(expense), 30, BLACK, true), -2);
    space(total, 8);
    add(total, label("Income  " + money(income), 14, 0xff786d42, false), -2);
    add(content, total, -2);
    space(content, 20);
    if (expense == 0) {
      empty(content, "A fresh month", "Your spending report will grow as you add expenses.", false);
      return;
    }
    Donut donut = new Donut(this, sums, expense);
    add(content, donut, 210);
    space(content, 12);
    add(content, label("Spending by Category", 18, BLACK, true), 36);
    for (String c : categories) {
      long amount = sums.get(c);
      if (amount == 0) continue;
      LinearLayout row = row(this);
      FrameLayout badge = new FrameLayout(this);
      badge.setBackground(bg(colorFor(c), 12, this));
      badge.addView(art(artFor(c)), fp(29, 29, Gravity.CENTER, 0, 0, 0, 0));

          label(
              money(amount) + "  ·  " + Math.round(100.0 * amount / expense) + "%",
              13,
              MUTED,
              false));
      add(content, row, 58);
      row.setOnClickListener(
          v -> {
            screen = "Transactions";
            categoryFilter = c;
            filter = "Expense";
            render();
          });
    }
    space(content, 12);
    TextView note =
        label(
            prefs.getBoolean("exclude_transfer", true)
                ? "Internal transfers are excluded from totals."
                : "Internal transfers are included in totals.",
            12,
            MUTED,
            false);
    add(content, note, -2);
  }

  private class Donut extends View {
    final Map<String, Long> sums;
    final long total;
    final Paint paint = new Paint(Paint.ANTI_ALIAS_FLAG);

    Donut(Context c, Map<String, Long> sums, long total) {
      super(c);
      this.sums = sums;
      this.total = total;
      setContentDescription("Expense breakdown by category");
    }

    protected void onDraw(Canvas canvas) {
      float cx = getWidth() / 2f, cy = getHeight() / 2f, r = d(77);
      RectF oval = new RectF(cx - r, cy - r, cx + r, cy + r);
      paint.setStyle(Paint.Style.STROKE);
      paint.setStrokeWidth(d(27));
      float start = -90;
      int i = 0;
      for (String c : categories) {
        float angle = 360f * sums.get(c) / total;
        paint.setColor(
            new int[] {0xffffb43d, 0xff49bcf1, 0xfff48fba, 0xff9c82fc, 0xff54d9b9, 0xffffd346}
                [i++]);
        if (angle > 0) canvas.drawArc(oval, start + 1, Math.max(1, angle - 2), false, paint);
        start += angle;
      }
      paint.setStyle(Paint.Style.FILL);
      paint.setColor(MUTED);
      paint.setTextAlign(Paint.Align.CENTER);
      paint.setTextSize(d(13));
      canvas.drawText("This month", cx, cy - d(6), paint);
      paint.setColor(BLACK);
      paint.setTypeface(Typeface.create("sans-serif-medium", Typeface.BOLD));
      paint.setTextSize(d(20));
      canvas.drawText(money(total), cx, cy + d(20), paint);
    }
  }
